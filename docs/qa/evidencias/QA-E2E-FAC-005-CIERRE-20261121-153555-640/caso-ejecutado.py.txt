def cerrar_alertas_y_logout(driver, wait):
    from selenium.common.exceptions import StaleElementReferenceException

    def libre(elemento):
        try:
            if not elemento.is_displayed() or not elemento.is_enabled():
                return False
            return driver.execute_script("""
                const e=arguments[0],r=e.getBoundingClientRect();
                const x=r.left+r.width/2,y=r.top+r.height/2;
                if(x<0 || y<0 || x>=innerWidth || y>=innerHeight) return false;
                const p=document.elementFromPoint(x,y);
                return p===e || e.contains(p);
            """,elemento)
        except StaleElementReferenceException:
            return False

    cerradas=0
    for _ in range(8):
        def accion(d):
            try:
                alertas=[a for a in d.find_elements(By.CSS_SELECTOR,".flash-container .alert.show") if a.is_displayed()]
                if not alertas:return ("sin_alertas",None,None)
                for alerta in reversed(alertas):
                    for boton in alerta.find_elements(By.CSS_SELECTOR,"button[data-bs-dismiss='alert']"):
                        if libre(boton):return ("cerrar",boton,alerta)
                return False
            except StaleElementReferenceException:
                return False
        tipo,boton,alerta=wait.until(accion)
        if tipo=="sin_alertas":break
        boton.click()
        wait.until(EC.invisibility_of_element(alerta))
        cerradas+=1
    else:
        raise AssertionError("Demasiadas alertas pendientes; no se fuerza el logout")

    def objetivo(d):
        botones=d.find_elements(By.CSS_SELECTOR,"form[action='/auth/logout'] button[type='submit']")
        return next((b for b in botones if libre(b)),False)
    wait.until(objetivo).click()
    return cerradas


def descargar_en_sesion_del_navegador(driver, url, ev):
    import base64
    actual=urlsplit(driver.current_url)
    destino=urlsplit(url)
    assert (destino.scheme,destino.netloc)==(actual.scheme,actual.netloc)
    driver.set_script_timeout(25)
    resultado=driver.execute_async_script("""
        const destino=new URL(arguments[0],location.href);
        const fin=arguments[arguments.length-1];
        if(destino.origin!==location.origin){fin({error:'origen_distinto'});return;}
        const control=new AbortController();
        const timer=setTimeout(()=>control.abort(),18000);
        fetch(destino.href,{method:'GET',credentials:'same-origin',redirect:'error',
            cache:'no-store',signal:control.signal})
        .then(async r=>{
            const a=new Uint8Array(await r.arrayBuffer());
            let bin='';
            for(let i=0;i<a.length;i+=8192)
                bin+=String.fromCharCode.apply(null,a.subarray(i,i+8192));
            fin({status:r.status,url:r.url,redirected:r.redirected,
                content_type:r.headers.get('Content-Type')||'',
                content_disposition:r.headers.get('Content-Disposition')||'',
                b64:btoa(bin)});
        }).catch(e=>fin({error:e.name})).finally(()=>clearTimeout(timer));
    """,url)
    metadatos={k:v for k,v in resultado.items() if k!='b64'}
    metadatos['cliente']='fetch mismo origen en Chrome; credenciales gestionadas por navegador'
    metadatos['proteccion_sesion']='sin cambios; no se exportan cookies'
    (ev/'respuesta-descarga-navegador.json').write_text(json.dumps(metadatos,indent=2),encoding='utf-8')
    assert not resultado.get('error'),metadatos
    assert resultado['status']==200 and resultado['url']==url and not resultado['redirected']
    tipo=resultado['content_type'];disposicion=resultado['content_disposition']
    assert 'xml' in tipo.lower() and 'attachment' in disposicion.lower()
    return base64.b64decode(resultado['b64'],validate=True),tipo,disposicion
